```ts
import { describe, expect, it } from 'vitest'
import { pickNextFestivals, type LandingEventRow } from './landing'

const TODAY = new Date(2026, 9, 14)
const OLD = '2026-01-01T10:00:00Z'

function row(id: string, start: string, over: Partial<LandingEventRow> = {}): LandingEventRow {
  return {
    id,
    name: `Festival ${id}`,
    image_url: `https://img/${id}.jpg`,
    tags: null,
    start_date: start,
    end_date: start,
    city: 'Limoges',
    department: '87',
    created_at: OLD,
    ...over,
  }
}

const none = new Map<string, number>()

describe('pickNextFestivals', () => {
  it('garde les cinq premières, dans l’ordre des dates', () => {
    const rows = ['a', 'b', 'c', 'd', 'e', 'f'].map((id, i) => row(id, `2026-11-0${i + 1}`))
    expect(pickNextFestivals(rows, none, TODAY).map((f) => f.id)).toEqual(['a', 'b', 'c', 'd', 'e'])

  it('préfère les dates qui ont une affiche, puis les remet dans l’ordre des dates', () => {
    const rows = [
      row('sans', '2026-11-01', { image_url: null }),
      ...['a', 'b', 'c', 'd', 'e'].map((id, i) => row(id, `2026-11-1${i}`)),

    expect(pickNextFestivals(rows, none, TODAY).map((f) => f.id)).toEqual(['a', 'b', 'c', 'd', 'e'])
  })

  it('complète avec des dates sans affiche s’il en manque', () => {
    const rows = [row('a', '2026-11-01'), row('sans', '2026-11-02', { image_url: null })]
    expect(pickNextFestivals(rows, none, TODAY).map((f) => f.id)).toEqual(['a', 'sans'])

  it('rend une liste vide sans date', () => {
    expect(pickNextFestivals([], none, TODAY)).toEqual([])
  })

  it('écrit le lieu « Ville (dép.) », ou la ville seule', () => {
    const [avec, sans] = pickNextFestivals(
      [row('a', '2026-11-01'), row('b', '2026-11-02', { department: '' })],
      none,
      TODAY,
    )
    expect(avec.place).toBe('Limoges (87)')
    expect(sans.place).toBe('Limoges')
  })

  it('« Populaire » va à la date la plus suivie, à partir de trois inscrits', () => {
    const rows = [row('a', '2026-11-01'), row('b', '2026-11-02')]
    const picked = pickNextFestivals(rows, new Map([['a', 2], ['b', 3]]), TODAY)
    expect(picked.map((f) => f.badge)).toEqual([null, 'Populaire'])

  it('pas de « Populaire » sous trois inscrits', () => {
    const picked = pickNextFestivals([row('a', '2026-11-01')], new Map([['a', 2]]), TODAY)
    expect(picked[0].badge).toBeNull()

    const rows = [row('a', '2026-11-01'), row('b', '2026-11-02')]
```

```ts
/**
 * QUOI     — les prochains festivals de la landing : lesquels, dans quel ordre, avec quelle
 *            pastille (« Populaire », « Nouveau »).
 * POURQUOI — lot 11a (spec 2026-10-14-v2-landing-design.md) : une vitrine se montre avec des
 *            affiches, donc les dates qui en ont passent d'abord ; on les montre ensuite dans
 *            l'ordre du calendrier.
 * ATTENTION — « Populaire » ne va qu'à UNE date, et seulement à partir de POPULAR_MIN inscrits :
 *            en dessous, la pastille mentirait.
 */
import { isRecent, parseSqlDate } from './dates'

export const NEXT_FESTIVALS = 5
export const POPULAR_MIN = 3
/** Combien de dates à venir on lit pour en choisir cinq. */
export const CANDIDATES = 20

export interface LandingEventRow {
  id: string
  name: string
  image_url: string | null
  tags: string[] | null
  start_date: string
  end_date: string
  city: string
  department: string | null
  created_at: string
}

export interface LandingFestival {
  id: string
  name: string
  imageUrl: string | null
  tags: string[] | null
  startDate: Date
  endDate: Date
  place: string
  badge: 'Populaire' | 'Nouveau' | null
}

/** `rows` arrive trié par date de début ; `registered` compte les inscrits par événement. */
export function pickNextFestivals(
  rows: LandingEventRow[],
  registered: ReadonlyMap<string, number>,
  today: Date,
): LandingFestival[] {
  const withPoster = rows.filter((r) => r.image_url)
  const withoutPoster = rows.filter((r) => !r.image_url)
  const picked = [...withPoster, ...withoutPoster]
    .slice(0, NEXT_FESTIVALS)
    .sort((a, b) => a.start_date.localeCompare(b.start_date))

  let popular: string | null = null
  let best = POPULAR_MIN - 1
  for (const r of picked) {
    const count = registered.get(r.id) ?? 0
    if (count > best) {
      best = count
      popular = r.id
    }
  }

  return picked.map((r) => ({
    id: r.id,
    name: r.name,
    imageUrl: r.image_url,
    tags: r.tags,
    startDate: parseSqlDate(r.start_date),
    endDate: parseSqlDate(r.end_date),
    place: r.department ? `${r.city} (${r.department})` : r.city,
    badge: r.id === popular ? 'Populaire' : isRecent(r.created_at, today) ? 'Nouveau' : null,
  }))
}
```

Le patron est celui de `features/login/useLoginWall.ts` (lu sans compte, vérifié le 11/10/2026 ; `participations` vérifié lisible sans compte le 14/10/2026). Les inscrits sont les statuts `inscrit` et `confirme` (même règle que `features/explorer/ExploreCard.tsx`).

```ts
/**
 * QUOI     — ce que la landing lit en base : les catégories du défilé et les prochains festivals.
 * POURQUOI — lot 11a : de vraies dates et de vraies catégories. Tout ici est lisible sans compte
 *            (events, tags, participations — vérifié le 14/10/2026).
 * ATTENTION — une lecture ratée rend `festivals: null` : la section se cache plutôt que de
 *            montrer une rangée vide.
 */
import { useEffect, useState } from 'react'
import { todayIso } from '@/lib/dates'
import { CANDIDATES, pickNextFestivals, type LandingFestival } from '@/lib/landing'
import { supabase } from '@/lib/supabase'
import { fetchTags, type TagRow } from '@/lib/tags'

interface Landing {
  tags: TagRow[]
  festivals: LandingFestival[] | null
}

async function loadFestivals(today: Date): Promise<LandingFestival[] | null> {
  const events = await supabase
    .from('events')
    .select('id, name, image_url, tags, start_date, end_date, city, department, created_at')
    .eq('is_private', false)
    .gte('end_date', todayIso(today))
    .order('start_date')
    .order('id')
    .limit(CANDIDATES)
  if (events.error) return null

  const registered = new Map<string, number>()
  if (events.data.length > 0) {
    const goers = await supabase
      .from('participations')
      .select('event_id')
      .in('event_id', events.data.map((e) => e.id))
      .in('status', ['inscrit', 'confirme'])
    // Sans les inscrits, la rangée s'affiche quand même, sans « Populaire ».
    if (!goers.error) {
      for (const g of goers.data) registered.set(g.event_id, (registered.get(g.event_id) ?? 0) + 1)
    }
  }
  return pickNextFestivals(events.data, registered, today)
}

export function useLanding(): Landing {
  const [landing, setLanding] = useState<Landing>({ tags: [], festivals: null })

  useEffect(() => {
    let cancelled = false
    async function run() {
      const today = new Date()
      const [tags, festivals] = await Promise.all([
        fetchTags(),
        loadFestivals(today).catch(() => null),
      ])
      if (cancelled) return
      setLanding({ tags, festivals })
    }
    void run()
    return () => {
      cancelled = true
    }
  }, [])

  return landing
}
```
