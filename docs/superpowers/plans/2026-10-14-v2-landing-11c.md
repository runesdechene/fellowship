**Produces :**
```ts
export interface ShowcaseRow { id: string; name: string; image_url: string | null; tags: string[] | null; start_date: string; end_date: string; city: string; department: string | null; created_at: string }
export interface ShowEvent { id: string; name: string; imageUrl: string | null; tags: string[] | null; startDate: Date; endDate: Date; place: string }
export type DemoStatus = 'interesse' | 'envoye' | 'inscrit'
export interface Showcase {
  wall: string[]                                   // affiches à venir, ≤ WALL_POSTERS
  crowd: { event: ShowEvent; count: number } | null // le festival à venir le plus suivi (≥ 1 inscrit)
  exhibitors: number | null
  months: { month: Date; events: (ShowEvent & { status: DemoStatus })[] }[] // 4 mois depuis aujourd'hui, ≤ 2 dates par mois
  dossiers: (ShowEvent & { status: DemoStatus; paid: number; total: number })[] // 4 premières dates
  oneClick: ShowEvent | null                        // une date à venir avec affiche, autre que crowd
  pdf: ShowEvent | null                             // la date qui titre le dossier PDF
  recent: (ShowEvent & { daysAgo: number })[]      // 3 derniers ajouts
  forYou: ShowEvent[]                               // 3 dates à venir avec affiche
}
export function buildShowcase(input: { upcoming: ShowcaseRow[]; registered: ReadonlyMap<string, number>; recent: ShowcaseRow[]; exhibitors: number | null }, today: Date): Showcase
```
